"""ops.py: budget ledger, watchdog, step planner, gating, metrics."""
from __future__ import annotations
import argparse, json, os, sys, time
from datetime import datetime, timedelta, timezone

LEDGER_NAME = "budget_ledger.json"
RATE = 1.98
BUDGET = 50.0
CAP_HOURS = 24.0

def _utcnow():
    return datetime.now(timezone.utc)

def parse_utc(s: str) -> datetime:
    dt = datetime.fromisoformat(s)
    if dt.tzinfo is None:
        raise ValueError("started-at must include timezone offset (UTC required)")
    return dt.astimezone(timezone.utc)

def load_ledger(run_dir: str) -> dict:
    path = os.path.join(run_dir, LEDGER_NAME)
    with open(path, "r", encoding="utf-8") as f:
        return json.load(f)

def deadline_of(ledger: dict):
    start = parse_utc(ledger["billing_started_at"])
    by_budget = start + timedelta(hours=(BUDGET - ledger["other_cost"]) / RATE)
    by_cap = start + timedelta(hours=CAP_HOURS)
    return min(by_budget, by_cap)

def cmd_init(args):
    os.makedirs(args.run_dir, exist_ok=True)
    path = os.path.join(args.run_dir, LEDGER_NAME)
    if os.path.exists(path):
        ledger = load_ledger(args.run_dir)
        print("existing ledger preserved:", ledger["billing_started_at"])
        return
    started = parse_utc(args.started_at)
    ledger = {
        "billing_started_at": started.isoformat(),
        "other_cost": float(args.other_cost),
        "run_dir": os.path.abspath(args.run_dir),
    }
    tmp = path + ".tmp"
    with open(tmp, "w", encoding="utf-8") as f:
        json.dump(ledger, f, indent=2)
    os.replace(tmp, path)
    print("deadline:", deadline_of(ledger).isoformat())

def shutdown_system(dry_run: bool) -> None:
    if dry_run:
        print("[dry-run] would invoke: shutdown -h now")
        return
    if sys.platform.startswith("win"):
        raise RuntimeError("real shutdown forbidden on Windows")
    os.system("shutdown -h now")

def cmd_watchdog(args):
    ledger = load_ledger(args.run_dir)
    dl = deadline_of(ledger)
    print("watchdog armed; deadline", dl.isoformat())
    while True:
        now = _utcnow()
        if now >= dl:
            shutdown_system(args.dry_run)
            if args.dry_run:
                print("[dry-run] deadline reached; no shutdown issued")
            return
        time.sleep(min(60, max(1, (dl - now).total_seconds())))

def choose_steps(train_seconds_per_step: float, eval_seconds: float,
                 overhead_seconds: float = 600.0) -> int:
    for T in range(4000, 999, -4):
        if 2 * T * train_seconds_per_step + 8 * eval_seconds + overhead_seconds <= 21600:
            return T
    raise ValueError("no feasible T in [1000,4000] (multiple of 4)")

def gate_first(a4, b4, start_miou: float) -> bool:
    maxA, maxB = max(a4), max(b4)
    if not (maxB - maxA >= 0.003):
        return False
    paired = sum(1 for x, y in zip(a4, b4) if y > x)
    if paired < 2:
        return False
    if maxB < start_miou:
        return False
    return True

def choose_checkpoint(points) -> int:
    best = None
    for step, miou in points:
        if best is None or miou > best[1]:
            best = (step, miou)
    return best[0]

def metrics_from_counts(counts) -> dict:
    """counts: {class: {tp,fp,fn}}; sums per class across images must be pre-aggregated."""
    out = {}
    for cls, c in counts.items():
        tp, fp, fn = c["tp"], c["fp"], c["fn"]
        denom = tp + fp + fn
        if denom == 0:
            raise ValueError(f"class {cls}: union is zero; rejecting")
        out[cls] = {"iou": tp / denom, "tp": tp, "fp": fp, "fn": fn}
    return out

def main(argv=None):
    p = argparse.ArgumentParser(prog="ops")
    sub = p.add_subparsers(dest="cmd", required=True)
    pi = sub.add_parser("init")
    pi.add_argument("--started-at", required=True)
    pi.add_argument("--other-cost", type=float, required=True)
    pi.add_argument("--run-dir", required=True)
    pi.set_defaults(func=cmd_init)
    pw = sub.add_parser("watchdog")
    pw.add_argument("--run-dir", required=True)
    pw.add_argument("--dry-run", action="store_true")
    pw.set_defaults(func=cmd_watchdog)
    args = p.parse_args(argv)
    args.func(args)

if __name__ == "__main__":
    main()

def sha256(path: str) -> str:
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()

def json_load_safe(path: str):

def compare_h5_h7(h5_path: str, h7_path: str) -> dict:
    a = json_load_safe(h5_path)
    b = json_load_safe(h7_path)
    result = {"h5": h5_path, "h7": h7_path, "numerically_equal": True, "details": []}
    if isinstance(a, dict) and isinstance(b, dict):
        for key in sorted(set(a) | set(b)):
            va, vb = a.get(key), b.get(key)
            if isinstance(va, (int, float)) and isinstance(vb, (int, float)):
                eq = abs(float(va) - float(vb)) <= 1e-9
            else:
                eq = va == vb
            if not eq:
                result["numerically_equal"] = False
            result["details"].append({"key": key, "equal": eq})
    else:
        result["numerically_equal"] = a == b
    return result

        out = {}
        for k, v in raw.items():

                out[str(k)] = [str(x) for x in v]
            elif isinstance(v, dict):
                ids = v.get("ids") or v.get("images") or v.get("files") or []
                out[str(k)] = [str(x) for x in ids]

    return {}
