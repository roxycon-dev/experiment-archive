def inject_repo(repo):
    repo = os.path.abspath(repo)
    if not os.path.isdir(repo):
        raise RuntimeError('repo directory does not exist: %s' % repo)
    os.chdir(repo)
    if repo not in sys.path:
        sys.path.insert(0, repo)
    return repo


def atomic_json(path, obj):
    tmp = path + '.tmp'
    with open(tmp, 'w', encoding='utf-8') as f:
        json.dump(obj, f, indent=2, ensure_ascii=False)
    f = None
    os.replace(tmp, path)


def load_config(repo, config_rel):
    """Load config, register custom imports and disable pretrained weights.
    Does NOT touch the model's reference/source_checkpoint setting here --
    the driver patches cfg.model['source_checkpoint'] to an absolute B path
    BEFORE the constructor."""
    from mmcv import Config
    cfg = Config.fromfile(os.path.join(repo, config_rel))
    model = cfg.get('model', None)
    if model is not None and 'pretrained' in model:
        model['pretrained'] = None
    _register_imports(cfg)
    return cfg


def _register_imports(cfg):
    imports = cfg.get('custom_imports', None)
    if imports:
        from mmseg.utils import register_custom_imports
        register_custom_imports(cfg)

def _load_state(path):
    """weights_only=False (trusted local file) with fallback for old torch
    versions that lack the kwarg."""
    try:
        ckpt = torch.load(path, map_location='cpu', weights_only=False)
    except TypeError:
        ckpt = torch.load(path, map_location='cpu')
    if not isinstance(ckpt, dict):
        raise RuntimeError('bad checkpoint (not a dict): %s' % path)
    state = ckpt.get('state_dict', ckpt)
    return {k[7:] if k.startswith('module.') else k: v
            for k, v in state.items()}


def _strict_load_full(model, path):
    """Full-state, strict load.  Any missing/unexpected key fails closed."""
    state = _load_state(path)
    model.load_state_dict(state, strict=True)
    return model


def snapshot_state(model):

def snapshot_buffers(model):

def _authorized_param(name):
    if name.startswith('reference.'):
        return False
    return name.startswith('decode_head') or name.startswith('backbone.stage4')


def first_update_evidence(model, ref_snapshot, bn_snapshot):
    """Verify after the first update:
    - every frozen parameter (reference AND any other non-allowed) unchanged
    - all BN buffers (running_mean/var, num_batches_tracked) unchanged
    - gradients present and nonzero on >=1 decode_head param AND >=1
      backbone.stage4 param, with finite total grad norm
    - at least one allowed (trainable) parameter actually changed
    Returns dict of booleans only."""
    ev = {'frozen_params_unchanged': True,
          'bn_buffers_unchanged': True,
          'grad_at_decoder': False,
          'grad_at_stage4': False,
          'grad_norm_finite': False,
          'allowed_changed': False}
    total_norm = 0.0
    for n, p in model.named_parameters():
        cur = p.detach().cpu()
        old = ref_snapshot.get(n)
        changed = old is not None and not torch.equal(cur, old)
        if _authorized_param(n):
            if changed:
                ev['allowed_changed'] = True
            if p.grad is not None:
                g = p.grad.detach().cpu()
                gn = float(torch.norm(g))
                if gn > 0:
                    total_norm += gn * gn
                    if n.startswith('decode_head'):
                        ev['grad_at_decoder'] = True
                    if n.startswith('backbone.stage4'):
                        ev['grad_at_stage4'] = True
        else:
            if changed:
                ev['frozen_params_unchanged'] = False
    ev['grad_norm_finite'] = bool(np.isfinite(total_norm) and total_norm > 0.0)
    for n, b in model.named_buffers():
        cur = b.detach().cpu()
        old = bn_snapshot.get(n)
        if old is not None and not torch.equal(cur, old):
            if ('running_mean' in n or 'running_var' in n or
                    n.endswith('num_batches_tracked')):
                ev['bn_buffers_unchanged'] = False
    return ev

def _digest_bytes(b):
    return hashlib.sha256(b).hexdigest()


def _arr_sha(a):
    return _digest_bytes(np.ascontiguousarray(a).tobytes())


def _unwrap_img(x):
    n = 0
    while hasattr(x, 'data') and n < 8:
        x = x.data
        n += 1
    while isinstance(x, (list, tuple)):
        x = x[0]
        while hasattr(x, 'data') and n < 16:
            x = x.data
            n += 1
    return x


def _unwrap_meta(x):
    """Original DataContainer unwrap."""
    n = 0
    while hasattr(x, 'data') and n < 8:
        x = x.data
        n += 1
    if isinstance(x, dict):
        return [x]
    out = []
    for e in x:
        e = e.data if hasattr(e, 'data') else e
        out.extend([e] if isinstance(e, dict) else [])
    return out


class ReferenceCache(object):
    VERSION = 'refcache-v2'

    def __init__(self, cache_dir, settings, expected_count=None):
        self.dir = os.path.abspath(cache_dir)
        os.makedirs(self.dir, exist_ok=True)
        self.manifest_path = os.path.join(self.dir, 'manifest.json')
        self.complete_path = os.path.join(self.dir, 'COMPLETE')
        self.settings = settings
        self._expected_count = expected_count  # test-only override
        self._manifest = None

    # -- identity material ------------------------------------------------
    def _cfg_digest(self):
        cfg = self.settings['_cfg']
        return _digest_bytes(cfg.dump().encode('utf-8'))

    def _train_image_ids(self):
        data_root = self.settings['data_root']
        lst = os.path.join(data_root, 'train.txt')
        if os.path.isfile(lst):
            ids = [os.path.splitext(os.path.basename(l.strip()))[0]
                   for l in open(lst, encoding='utf-8') if l.strip()]
        else:
            d = os.path.join(data_root, 'images', 'train')
            ids = [os.path.splitext(f)[0] for f in sorted(os.listdir(d))
                   if f.lower().endswith(('.jpg', '.jpeg'))]
        return sorted(ids)

    def _source_images_digest(self):
        """Digest over the full train image set (name+file sha), full split
        IDs -- never only dict keys."""
        data_root = self.settings['data_root']
        d = os.path.join(data_root, 'images', 'train')
        h = hashlib.sha256()
        for f in sorted(os.listdir(d)):
            if f.lower().endswith(('.jpg', '.jpeg')):
                h.update(f.encode('utf-8'))
                h.update(sha256(os.path.join(d, f)).encode('utf-8'))
        return h.hexdigest()

    def _ref_weight_sha(self):
        return self.settings['_hashes']['reference_checkpoint']

    # -- completeness ------------------------------------------------------
    def complete(self):
        """Missing cache -> False (clean).  Present but inconsistent ->
        RuntimeError (corruption, fail closed).  Validates: version, split
        == train with exact full ID list, source image digest, reference
        weight sha (NOT student), effective config digest, normalization,
        per-entry file hash / shape / finite / range."""
        if not (os.path.isfile(self.manifest_path) and
                os.path.isfile(self.complete_path)):
            return False
        try:
            with open(self.manifest_path, 'r', encoding='utf-8') as f:
                man = json.load(f)
        except (IOError, ValueError):
            raise RuntimeError('reference cache manifest unreadable')
        self._fail_if(man.get('version') != self.VERSION, 'version')
        self._fail_if(man.get('split') != 'train', 'split')
        self._fail_if(sorted(man.get('image_ids', [])) !=
                      self._train_image_ids(), 'train image IDs')
        self._fail_if(man.get('source_images_sha256') !=
                      self._source_images_digest(), 'source images')
        self._fail_if(man.get('reference_weight_sha256') !=
                      self._ref_weight_sha(), 'reference weight')
        self._fail_if(man.get('config_sha256') != self._cfg_digest(),
                      'config digest')
        self._fail_if(man.get('normalization') !=
                      {'mean': IMG_MEAN.tolist(), 'std': IMG_STD.tolist()},
                      'normalization')
        entries = man.get('entries', {})
        if not entries:
            raise RuntimeError('reference cache corrupt: no entries')
        for stem, e in sorted(entries.items()):
            npy = e.get('npy')
            self._fail_if(not npy or not os.path.isfile(npy),
                          'missing file %s' % stem)
            self._fail_if(sha256(npy) != e.get('sha256'),
                          'file hash %s' % stem)
            self._fail_if(not npy.endswith('.npy'), 'format %s' % stem)
        self._fail_if(self._expected_count is not None and
                      len(entries) != self._expected_count, 'entry count')
        self._manifest = man
        return True

    @staticmethod
    def _fail_if(cond, what):
        if cond:
            raise RuntimeError('reference cache corrupt: %s mismatch' % what)

    # -- access ------------------------------------------------------------
    def get(self, stem):
        """Lazy load of one entry, hash + shape + finite/range validated."""
        if self._manifest is None:
            if not self.complete():
                raise RuntimeError('reference cache incomplete or corrupt')
        e = self._manifest['entries'][stem]
        npy = e['npy']
        if sha256(npy) != e['sha256']:
            raise RuntimeError('cache entry hash mismatch: %s' % stem)
        arr = np.load(npy)
        if arr.dtype != np.float32 or arr.ndim != 3 or arr.shape[2] != 4:
            raise RuntimeError('bad cache entry %s' % stem)
        if list(arr.shape) != list(e['shape']):
            raise RuntimeError('cache shape mismatch: %s' % stem)
        if not np.isfinite(arr).all() or arr.min() < 0.0 or arr.max() > 1.0:
            raise RuntimeError('cache entry out of range: %s' % stem)
        return arr

CLASSES = ['EX', 'HE', 'SE', 'MA']


def per_image_counts(prob, gt):
    """prob (4,H,W) float, gt (H,W) int with 255 ignore.  Threshold 0.5
    strict >, background (class 0) not counted."""
    counts = []
    for c in range(4):
        p = prob[c] > 0.5
        g = gt == (c + 1)
        ig = gt == 255
        tp = int(np.logical_and(p, g).sum())
        fp = int(np.logical_and(p, np.logical_and(~g, ~ig)).sum())
        fn = int(np.logical_and(~p, g).sum())
        counts.append([tp, fp, fn])
    return counts


def global_miou(all_counts):
    ious = []
    per = {}
    for c, (tp, fp, fn) in enumerate(all_counts):
        union = tp + fp + fn
        if union == 0:
            raise RuntimeError('IoU union is zero for class %d' % (c + 1))
        iou = tp / union
        ious.append(iou)
        per[CLASSES[c]] = iou
    return float(np.mean(ious)), per


def evaluate_split(predict_fn, split, data_root, cfg, deadline, log,
                   dump_counts=None, expected_count=None,
                   _ds_builder=None):  # _ds_builder: test-only injection
    """Original val/test dataset pipeline; gt nearest-resized to ori_shape.
    Predictor may return (1,4,H,W) or (4,H,W) -- only the batch dim is
    squeezed; shape/finite/range validated strictly.  The caller owns
    model.eval().  Returns (miou, per_class, per_image_counts)."""
    import cv2
    if _ds_builder is None:
        from mmseg.datasets import build_dataset
        d = copy.deepcopy(cfg.data[split])
        d.test_mode = True
        d.img_dir = os.path.join(data_root, 'images', split)
        d.ann_dir = os.path.join(data_root, 'labels', split)
        ds = build_dataset(d)
    else:
        ds = _ds_builder(data_root, split)
    n = len(ds)
    expected = expected_count if expected_count is not None \
        else {'val': 76, 'test': 227}[split]
    if n != expected:
        raise RuntimeError('split %s has %d items, expected %d' %
                           (split, n, expected))
    gt_maps = ds.get_gt_seg_maps()
    total = [[0, 0, 0] for _ in range(4)]
    per_image = {}
    t0 = time.time()
    for i in range(n):
        if time.time() > deadline:
            raise RuntimeError('deadline hit during %s eval' % split)
        s = ds[i]
        img = _unwrap_img(s['img']).unsqueeze(0).cuda()
        meta = _unwrap_meta(s['img_metas'])
        prob = predict_fn(img, meta)
        if torch.is_tensor(prob):
            prob = prob.detach().cpu().numpy()
        prob = np.asarray(prob, dtype=np.float32)
        if prob.ndim == 4:
            if prob.shape[0] != 1:
                raise RuntimeError('predictor returned batch >1')
            prob = prob[0]                      # squeeze BATCH only
        if prob.ndim != 3 or prob.shape[0] != 4:
            raise RuntimeError('predictor must return 4HW, got %s' %
                               (prob.shape,))
        ori = tuple(meta[0]['ori_shape'][:2])
        if prob.shape[1:] != ori:
            raise RuntimeError('prob shape %s != ori_shape %s' %
                               (prob.shape[1:], ori))
        if not np.isfinite(prob).all() or prob.min() < 0.0 or prob.max() > 1.0:
            raise RuntimeError('predictor output not finite prob in [0,1]')
        gt = np.asarray(gt_maps[i]).astype(np.int64)
        if gt.shape != ori:
            gt = cv2.resize(gt.astype(np.uint8), (ori[1], ori[0]),
                            interpolation=cv2.INTER_NEAREST).astype(np.int64)
        if not np.all(np.isin(gt, (0, 1, 2, 3, 4, 255))):
            raise RuntimeError('gt contains invalid labels outside 0..4/255')
        cnt = per_image_counts(prob, gt)
        name = os.path.splitext(os.path.basename(meta[0]['ori_filename']))[0]
        per_image[name] = {CLASSES[c]: cnt[c] for c in range(4)}
        for c in range(4):
            for j in range(3):
                total[c][j] += cnt[c][j]
        if (i + 1) % 25 == 0:
            log('eval %s %d/%d (%.1fs)' % (split, i + 1, n, time.time() - t0))
    miou, per = global_miou([tuple(x) for x in total])
    if dump_counts:
        d = os.path.dirname(dump_counts)
        if d:
            os.makedirs(d, exist_ok=True)
        atomic_json(dump_counts, {'split': split, 'miou': miou,
                                  'per_class': per, 'global_counts': total,
                                  'per_image': per_image})
    return miou, per, per_image

class CorrectionHead(nn.Module):
    """7 -> 16 -> 16 -> 1 conv head, output scaled by 6*tanh."""
    def __init__(self):
        super(CorrectionHead, self).__init__()
        self.net = nn.Sequential(
            nn.Conv2d(7, 16, 3, 1, 1), nn.ReLU(inplace=True),
            nn.Conv2d(16, 16, 3, 1, 1), nn.ReLU(inplace=True),
            nn.Conv2d(16, 1, 1))

    def forward(self, f):
        return 6.0 * torch.tanh(self.net(f))


def load_flat_head(head_state, device='cpu'):
    """Load the ONLY correction head (MA) from h4_best.ck['head'], a flat
    state_dict with keys net.0.weight, net.0.bias, ...  Fail closed on
    anything else."""
    if not isinstance(head_state, dict) or 'net.0.weight' not in head_state:
        raise RuntimeError('champion head file: expected flat state dict '
                           'with net.* keys')
    h = CorrectionHead().to(device)
    h.load_state_dict(head_state, strict=True)
    h.eval()
    for p in h.parameters():
        p.requires_grad_(False)
    return h


def _logit(p, eps=1e-4):
    q = p.clamp(eps, 1 - eps)
    return torch.log(q / (1 - q))


def member_prob(model, img, meta, flip, device):
    """Exact member inference via model(return_loss=False, rescale=True).
    For flip: pass the flipped image and flip flag in img_metas -- the
    mmseg test pipeline already un-flips the output, so the returned
    probability is NOT flipped a second time here."""
    meta_f = copy.deepcopy(meta)
    x = img
    if flip:
        meta_f[0]['flip'] = True
        meta_f[0]['flip_direction'] = 'horizontal'
        x = img.flip(-1)
    out = model(return_loss=False, rescale=True, img=[x], img_metas=[meta_f])
    out = out[0] if isinstance(out, (tuple, list)) else out
    p = torch.from_numpy(np.asarray(out, dtype=np.float32)).to(device)
    return p


class ChampionEnsemble(object):
    """EXACT champion: 4 named members D,E,B,E4 with weights 3,2,1,1,
    horizontal-flip TTA for the ensemble average, per-class MA correction
    head only (no HE/SE/MA head-set requirement), driven by ps/pr from the
    same D member without flip."""

    MEMBER_WEIGHTS = {'D': 3.0, 'E': 2.0, 'B': 1.0, 'E4': 1.0}

    def __init__(self, settings, champion_cfg, device='cuda'):
        from mmseg.models import build_segmentor
        self.device = device
        members = champion_cfg['members']
        if len(members) != 4:
            raise RuntimeError('champion must have exactly 4 members')
        names = sorted(str(s['name']) for s in members)
        if names != sorted(self.MEMBER_WEIGHTS):
            raise RuntimeError('champion members must be D,E,B,E4; got %s' %
                               names)
        self.members = []
        self.d_model = None
        for spec in members:
            name = spec['name']
            c = load_config(settings['repo'], spec['config'])
            # explicit source_checkpoint: settings reference path (B) unless
            # the spec explicitly provides one (e.g. E4 base read source).
            c.model['source_checkpoint'] = spec.get(
                'source_checkpoint', settings['reference_checkpoint'])
            m = build_segmentor(c.model)
            mode = spec.get('load', 'full')
            if mode == 'student_only':
                # explicit, configured student-only semantic (e.g. D), loaded
                # via core strict_initialize against the settings reference B.
                strict_initialize(m, spec['checkpoint'],
                                  settings['reference_checkpoint'])
            elif mode == 'full':
                _strict_load_full(m, spec['checkpoint'])
            else:
                raise RuntimeError('unknown load mode %r for %s' %
                                   (mode, name))
            m = m.to(device).eval()
            for p in m.parameters():
                p.requires_grad_(False)
            self.members.append((m, self.MEMBER_WEIGHTS[name], spec))
            if name == 'D':
                self.d_model = m
        if self.d_model is None:
            raise RuntimeError('champion requires a D member for ps/pr')
        # exactly one head: MA, loaded from h4_best.ck['head'] flat state.
        try:
            ck = torch.load(champion_cfg['head'], map_location='cpu',
                            weights_only=False)
        except TypeError:
            ck = torch.load(champion_cfg['head'], map_location='cpu')
        head_state = ck['head'] if isinstance(ck, dict) and 'head' in ck else ck
        self.heads = {'MA': load_flat_head(head_state, device)}

    def predict(self, img, meta):
        """Returns (4,H,W) corrected champion probability at ori resolution."""
        with torch.no_grad():
            acc = None
            tot = 0.0
            for m, w, spec in self.members:
                p = member_prob(m, img, meta, False, self.device)
                p2 = member_prob(m, img, meta, True, self.device)
                acc = (p + p2) * w if acc is None else acc + (p + p2) * w
                tot += 2.0 * w
            pf = acc / tot
            # same D member for ps/pr, no flip
            ps = self._raw_prob(self.d_model, img, meta, 'student')
            pr = self._raw_prob(self.d_model, img, meta, 'reference')
            ori = tuple(meta[0]['ori_shape'][:2])
            img_f = F.interpolate(img, size=ori, mode='bilinear',
                                  align_corners=False)[0]
            chans = []
            for c in range(4):
                cname = CLASSES[c]
                if cname in self.heads:
                    h = self.heads[cname]
                    f7 = torch.cat([img_f, _logit(pf[c:c + 1]),
                                    _logit(ps[c:c + 1]), _logit(pr[c:c + 1]),
                                    ps[c:c + 1] - pr[c:c + 1]],
                                   dim=0).unsqueeze(0)
                    newp = torch.sigmoid(_logit(pf[c:c + 1][0]) +
                                         h(f7)[0, 0])
                    chans.append(newp.unsqueeze(0))
                else:
                    chans.append(pf[c:c + 1])
            return torch.cat(chans, dim=0)

    def _raw_prob(self, m, img, meta, which):
        fn = getattr(m, '_' + which + '_prob', None)
        if fn is None:
            raise RuntimeError('D member lacks _%s_prob' % which)
        return fn(img, meta, True)[0].to(self.device)


def mix_probs(p_champ, p_new):
    return 0.5 * p_champ + 0.5 * p_new

def audit_split(data_root, expected=(454, 76, 227)):
    """Verify per-split image counts, no dup/intersection, txt IDs agree
    with image dirs AND png masks directly, and return the real assignment
    dict filename -> {'split', 'stem'} covering the full ID set."""
    assignment = {}
    for si, split in enumerate(('train', 'val', 'test')):
        img_dir = os.path.join(data_root, 'images', split)
        lab_dir = os.path.join(data_root, 'labels', split)
        list_file = os.path.join(data_root, split + '.txt')
        imgs = sorted(f for f in os.listdir(img_dir)
                      if f.lower().endswith(('.jpg', '.jpeg')))
        if len(imgs) != expected[si]:
            raise RuntimeError('split %s has %d images, expected %d' %
                               (split, len(imgs), expected[si]))
        if os.path.isfile(list_file):
            listed = [l.strip() for l in open(list_file, encoding='utf-8')
                      if l.strip()]
            if sorted(os.path.splitext(x)[0] for x in listed) != \
                    sorted(os.path.splitext(x)[0] for x in imgs):
                raise RuntimeError('%s does not match %s images' %
                                   (list_file, split))
        else:
            raise RuntimeError('missing split list %s' % list_file)
        stems = set()
        for f in imgs:
            stem = os.path.splitext(f)[0]
            if stem in assignment:
                raise RuntimeError('duplicate stem: %s' % stem)
            stems.add(stem)
            assignment[f] = {'split': split, 'stem': stem}
        labs = set(os.path.splitext(x)[0] for x in os.listdir(lab_dir)
                   if x.endswith('.png'))
        if split == 'test':
            if labs:
                raise RuntimeError('test labels present — forbidden')
        else:
            if labs != stems:
                raise RuntimeError('labels/%s mismatch' % split)
    return assignment

def parse_deadline(raw):
    """Accepts ISO-8601 string or epoch seconds; anything else rejected."""
    if isinstance(raw, (int, float)):
        return float(raw)
    if isinstance(raw, str):
        try:
            from datetime import datetime
            d = datetime.fromisoformat(raw.replace('Z', '+00:00'))
            if d.tzinfo is None:
                d = d.astimezone()
            return d.timestamp()
        except ValueError:
            pass
    raise ValueError('unknown deadline format (need ISO-8601 or epoch): %r'
                     % raw)

def validate_settings(path):
    with open(path, 'r', encoding='utf-8') as f:
        s = json.load(f)
    missing = [k for k in REQUIRED_KEYS if k not in s]
    if missing:
        raise RuntimeError('settings missing keys: %s' % missing)
    targets = {
        'config': s['config'],
        'student_checkpoint': s['student_checkpoint'],
        'reference_checkpoint': s['reference_checkpoint'],
        'champion_head': s['champion']['head'],
    }
    for m in s['champion']['members']:
        targets['member_%s_cfg' % m['name']] = m['config']
        targets['member_%s_ckpt' % m['name']] = m['checkpoint']
    hashes, pending = {}, []
    for key, rel in sorted(targets.items()):
        full = _abs(s['repo'], rel)
        if not os.path.isfile(full):
            pending.append(full)
            continue
        exp = s['expected_sha256'].get(rel)
        if exp is None:
            raise RuntimeError('no expected_sha256 for %s; no unhashed '
                               'fallback (fail closed)' % rel)
        if exp != sha256(full):
            raise RuntimeError('sha256 mismatch for %s' % rel)
        hashes[key] = exp
    s['_hashes'] = hashes
    s['_abs'] = {k: _abs(s['repo'], v) for k, v in targets.items()}
    s['_pending'] = pending
    return s


def sha256(path):
    h = hashlib.sha256()
    with open(path, 'rb') as f:
        for chunk in iter(lambda: f.read(1 << 20), b''):
            h.update(chunk)
    return h.hexdigest()


def _resolve_obj(o, repo):
    """Recursively resolve an mmcv config to JSON-safe primitives."""
    if isinstance(o, dict):
        return {str(k): _resolve_obj(v, repo) for k, v in o.items()}
    if isinstance(o, (list, tuple)):
        return [_resolve_obj(v, repo) for v in o]
    if isinstance(o, (str, int, float, bool)) or o is None:
        return o
    return repr(o)


def config_fingerprint(settings):
    """Fingerprint = recursive resolved cfg + ALL referenced source files."""
    import runtime as rt
    cfg = rt.load_config(settings['repo'], settings['config'])
    d = cfg.dump() if hasattr(cfg, 'dump') else json.dumps(
        _resolve_obj(cfg._cfg_dict if hasattr(cfg, '_cfg_dict') else {}, settings['repo']))
    blob = {'config_text_sha256': hashlib.sha256(d.encode()).hexdigest()}
    srcs = set()
    def walk(o):
        if isinstance(o, dict):
            for v in o.values():
                walk(v)
        elif isinstance(o, (list, tuple)):
            for v in o:
                walk(v)
        elif isinstance(o, str) and o.endswith('.py'):
            full = _abs(settings['repo'], o)
            if os.path.isfile(full):
                srcs.add(full)
    walk(_resolve_obj(cfg._cfg_dict if hasattr(cfg, '_cfg_dict') else {}, settings['repo']))
    blob['source_files'] = {}
    for p in sorted(srcs):
        blob['source_files'][p] = sha256(p)
    return blob

def load_protocol_state(run_dir, settings):

def phase_deadline(st, phase):
    """Absolute deadline of a phase, from persisted starts; capped by the
    global deadline derived from the persisted run start."""
    idx = PHASE_ORDER.index(phase)
    base = st['run_start']
    for p in PHASE_ORDER[:idx]:
        if p not in st['phase_start']:
            raise RuntimeError('phase %s not started' % p)
        base = max(base, st['phase_start'][p] + PHASE_CAPS[p])
    start = st['phase_start'].get(phase, base)
    return min(start + PHASE_CAPS[phase], st['run_start'] + GLOBAL_BUDGET)


def mark_phase(run_dir, st, phase):
    if phase not in st['phase_start']:
        st['phase_start'][phase] = time.time()
        save_protocol_state(run_dir, st)
    return phase_deadline(st, phase)

def audit_labels(data_root, split, assignment, log):
    """Verify each label png: readable, shape == image, values in 0..4/255.
    Streams file-by-file; nothing is kept in RAM."""
    import cv2
    lab_dir = os.path.join(data_root, 'labels', split)
    img_dir = os.path.join(data_root, 'images', split)
    for fname in sorted(assignment):
        stem = os.path.splitext(fname)[0]
        lp = os.path.join(lab_dir, stem + '.png')
        img = cv2.imread(os.path.join(img_dir, fname), 1)
        if img is None:
            raise RuntimeError('unreadable image: %s' % fname)
        lab = cv2.imread(lp, 0)
        if lab is None:
            raise RuntimeError('unreadable label: %s' % lp)
        if lab.shape[:2] != img.shape[:2]:
            raise RuntimeError('label/image shape mismatch: %s' % fname)
        u = np.unique(lab)
        ok = np.all((u == 255) | ((u >= 0) & (u <= 4)))
        if not ok:
            raise RuntimeError('label values out of range 0..4/255: %s' % lp)
    log('labels OK for %s (%d files)' % (split, len(assignment)))

def train_arm(settings, run_dir, arm, seed, T, deadline, log, ref_cache,
              split_hashes):
    import runtime as rt
    seed_everything(seed)          # BEFORE construction
    model, _ = build_student(settings)
    model.train()
    rt.freeze_bn(model)
    opt = torch.optim.SGD([p for n, p in model.named_parameters()
                           if p.requires_grad],
                          lr=5e-5, momentum=0.9, weight_decay=0.0005)
    ckpt_dir = os.path.join(run_dir, 'ckpt_%s_s%d' % (arm, seed))
    os.makedirs(ckpt_dir, exist_ok=True)
    last_path = os.path.join(ckpt_dir, 'last.pth')
    start, hist, gstate, ostate = 0, [], None, None
    if os.path.isfile(last_path):
        start, hist, gstate, ostate = resume_checkpoint(
            last_path, model, opt, seed, T, arm, split_hashes)
        log('%s s%d resumed at iter %d with %d val entries' %
            (arm, seed, start, len(hist)))
    geom_rng = np.random.RandomState(seed)
    order_rng = np.random.RandomState(seed)
    if gstate is not None:                       # restore LOCAL streams
        geom_rng.set_state(gstate)
        order_rng.set_state(ostate)
    data_root = settings['data_root']
    files = list_split(data_root, 'train')
    stems = [os.path.splitext(f)[0] for f in files]
    trace_path = os.path.join(run_dir, 'trace_%s_s%d.json' % (arm, seed))
    snap, bn_snap, checked = None, None, start > 0
    if not checked:
        snap, bn_snap = rt.snapshot_state(model), rt.snapshot_buffers(model)
    quarters = [T * q // 4 for q in (1, 2, 3, 4)]
    for it in range(start, T):
        if time.time() > deadline:
            raise ShutdownGraceful('deadline during %s s%d' % (arm, seed))
        model.train()
        rt.freeze_bn(model)
        opt.zero_grad()
        idxs = order_rng.permutation(len(stems))[:2]   # batch of 2
        imgs, labs, refs = [], [], []
        for i in idxs:
            stem = stems[i]
            assert 0 <= i < len(stems) and stem in ref_cache
            img, lab = rt.load_image_label(data_root, 'train', stem)
            r = ref_cache[stem]
            im2, la2, rf2, tr = rt.paired_geometry(img, lab, r, geom_rng)
            imgs.append(torch.from_numpy(im2))
            labs.append(torch.from_numpy(la2))
            refs.append(torch.from_numpy(rf2))
            if not os.path.isfile(trace_path):         # first-batch proof
                rt.atomic_json(trace_path, {'arm': arm, 'seed': seed,
                                            'iter': it, 'batches': [tr]})
        img_b = torch.stack(imgs).cuda()
        lab_b = torch.stack(labs).cuda()
        ref_b = torch.stack(refs).cuda()
        logits = model.encode_decode(img_b, [{}])
        loss = rt.loss_pair(logits, lab_b, ref_b, arm)
        loss.backward()
        torch.nn.utils.clip_grad_norm_(
            [p for p in model.parameters() if p.requires_grad], 5.0)
        opt.step()
        rt.freeze_bn(model)
        if not checked:
            ev = rt.first_update_evidence(model, snap, bn_snap)
            if not all(ev.values()):
                raise RuntimeError('first update violated: %s' % ev)
            rt.atomic_json(os.path.join(run_dir,
                                        'first_update_%s_s%d.json' % (arm, seed)), ev)
            del snap, bn_snap                 # free after pass
            snap = bn_snap = None
            checked = True
        if (it + 1) % 100 == 0 or (it + 1) == T:
            save_checkpoint(last_path, model, opt, it + 1, seed, T, arm,
                            settings, split_hashes, geom_rng, order_rng, hist)
        if (it + 1) in quarters:
            st = get_rng_state()
            was_training = model.training
            model.eval()                      # eval mode under no_grad
            with torch.no_grad():
                miou, _, _ = evaluate_split_local(
                    lambda i_, m_: d_inf(model, i_, m_), 'val', settings, log,
                    deadline)
            model.train(was_training)         # back to train, then freeze BN
            rt.freeze_bn(model)
            set_rng_state(st)                 # restore ALL rng
            hist.append((it + 1, miou))
            save_checkpoint(os.path.join(ckpt_dir, 'iter_%d.pth' % (it + 1)),
                            model, opt, it + 1, seed, T, arm, settings,
                            split_hashes, geom_rng, order_rng, hist)  # AFTER val
            log('%s s%d iter %d val mIoU %.4f' % (arm, seed, it + 1, miou))
    if len(hist) != 4:
        raise RuntimeError('expected 4 validation timepoints, got %d' % len(hist))
    best = rt.choose_checkpoint(hist)         # tie -> earlier step
    del model, opt
    torch.cuda.empty_cache()
    return best, hist


def d_inf(model, img, meta):
    """D-model inference; squeeze the batch dim (batch of 1 only)."""
    out = model.inference(img, meta, True)
    if isinstance(out, (tuple, list)):
        out = out[0]
    return out[0] if out.dim() == 4 else out     # (4,H,W)

def evaluate_candidates(settings, champion, new_model, split, deadline, log,
                        dump_prefix=None):
    """Shared dataset loop; champion probs + new probs computed once per
    image; counts for champion/new/mix accumulate from the SAME pass."""
    import runtime as rt
    import cv2
    from mmseg.datasets import build_dataset
    cfg = settings['_cfg']
    d = copy.deepcopy(cfg.data[split]) if False else None
    # local import guard for py2-style copy
    import copy as _c
    d = _c.deepcopy(cfg.data[split])
    d.test_mode = True
    d.img_dir = os.path.join(settings['data_root'], 'images', split)
    d.ann_dir = os.path.join(settings['data_root'], 'labels', split)
    ds = build_dataset(d)
    gt_maps = ds.get_gt_seg_maps()
    totals = {'champion': [[0, 0, 0] for _ in range(4)],
              'new': [[0, 0, 0] for _ in range(4)],
              'mix': [[0, 0, 0] for _ in range(4)]}
    for i in range(len(ds)):
        if time.time() > deadline:
            raise ShutdownGraceful('deadline during %s candidate eval' % split)
        s = ds[i]
        img = rt._unwrap_img(s['img']).unsqueeze(0).cuda()
        meta = rt._unwrap_meta(s['img_metas'])
        with torch.no_grad():
            pc = champion.predict(img, meta)                  # champion
            pn = d_inf(new_model, img, meta).detach()         # new (train=False)
            pm = rt.mix_probs(pc, pn)
        gt = np.asarray(gt_maps[i])
        ori = meta[0]['ori_shape'][:2]
        if gt.shape != tuple(ori):
            gt = cv2.resize(gt.astype(np.uint8), (ori[1], ori[0]),
                            interpolation=cv2.INTER_NEAREST)
        for name, p in (('champion', pc), ('new', pn), ('mix', pm)):
            cnt = rt.per_image_counts(
                p.detach().cpu().numpy().astype(np.float32), gt.astype(np.int64))
            for c in range(4):
                for j in range(3):
                    totals[name][c][j] += cnt[c][j]
    out = {}
    for name in totals:
        miou, per = rt.global_miou([tuple(x) for x in totals[name]])
        out[name] = {'miou': miou, 'per_class': per}
        log('final-candidates %s %s mIoU %.4f' % (split, name, miou))
    return out


# ---------------------------------------------------------------------------
# timing pilot
# ---------------------------------------------------------------------------
def pilot_timing(settings, run_dir, deadline, log, ref_cache,
                 reuse_eval_seconds=None):
    """50 updates per arm; proper cuda.synchronize; wall time includes data
    prep and checkpoint overhead; 10% margin applied by caller.  Does not
    mutate the run identity, no eval RNG side effects."""
    import runtime as rt
    max_step = 0.0
    for arm in ('A', 'B'):
        seed_everything(1000 + (0 if arm == 'A' else 1))
        model, _ = build_student(settings)
        model.train()
        rt.freeze_bn(model)
        opt = torch.optim.SGD([p for p in model.parameters() if p.requires_grad],
                              lr=5e-5, momentum=0.9, weight_decay=0.0005)
        rng = np.random.RandomState(42)
        files = list_split(settings['data_root'], 'train')
        stems = [os.path.splitext(f)[0] for f in files]
        for it in range(50):
            if time.time() > deadline:
                raise ShutdownGraceful('deadline during pilot')
            t0 = time.time()
            idxs = rng.permutation(len(stems))[:2]
            imgs, labs, refs = [], [], []
            for i in idxs:
                stem = stems[i]
                img, lab = rt.load_image_label(settings['data_root'], 'train', stem)
                im2, la2, rf2, _ = rt.paired_geometry(img, lab, ref_cache[stem], rng)
                imgs.append(torch.from_numpy(im2))
                labs.append(torch.from_numpy(la2))
                refs.append(torch.from_numpy(rf2))
            opt.zero_grad()
            logits = model.encode_decode(torch.stack(imgs).cuda(), [{}])
            loss = rt.loss_pair(logits, torch.stack(labs).cuda(),
                                torch.stack(refs).cuda(), arm)
            loss.backward()
            torch.nn.utils.clip_grad_norm_(
                [p for p in model.parameters() if p.requires_grad], 5.0)
            opt.step()
            if it == 49:
                save_checkpoint(os.path.join(
                    run_dir, 'pilot_%s.pth' % arm), model, opt, 50, 0, 50,
                    arm, settings, {}, rng, rng, [])   # overhead included
            torch.cuda.synchronize()
            max_step = max(max_step, time.time() - t0)
        del model, opt
        torch.cuda.empty_cache()
    t_eval = reuse_eval_seconds
    if t_eval is None:
        model, _ = build_student(settings)
        model.eval()
        t0 = time.time()
        with torch.no_grad():
            evaluate_split_local(lambda i_, m_: d_inf(model, i_, m_),
                                 'val', settings, log, deadline)
        t_eval = time.time() - t0
        del model
        torch.cuda.empty_cache()
    return max_step, t_eval


# ---------------------------------------------------------------------------
# selection lock (immutable) and test markers
# ---------------------------------------------------------------------------
def write_lock(run_dir, lock):
    path = os.path.join(run_dir, 'selection_lock.json')
    if os.path.isfile(path):
        with open(path, 'r', encoding='utf-8') as f:
            old = json.load(f)
        if old != lock:
            raise RuntimeError('selection_lock.json exists and differs; '
                               'locks are immutable')
        return path
    rt_like_atomic(path, lock)
    return path


def rt_like_atomic(path, obj):
    tmp = path + '.tmp'
    with open(tmp, 'w', encoding='utf-8') as f:
        json.dump(obj, f, indent=2, sort_keys=True)
    os.replace(tmp, path)


# ---------------------------------------------------------------------------
# main
# ---------------------------------------------------------------------------
def main():
    ap = argparse.ArgumentParser()
    ap.add_argument('--settings', required=True)
    ap.add_argument('--run-dir', required=True)
    ap.add_argument('--dry-run', action='store_true')
    args = ap.parse_args()

    settings = validate_settings(args.settings)
    os.makedirs(args.run_dir, exist_ok=True)
    log = Log(args.run_dir)

    # ---- dry-run: no mmcv / GPU / ledger needed; no fake pass ----
    if args.dry_run:
        pending = list(settings.get('_pending', []))
        for split in ('train', 'val', 'test'):
            d = os.path.join(settings['data_root'], 'images', split)
            if os.path.isdir(d):
                n = len(list_split(settings['data_root'], split))
                log('dry-run: %s images = %d' % (split, n))
            else:
                pending.append(d)
        for ph in ('train', 'val'):
            d = os.path.join(settings['data_root'], 'labels', ph)
            if not os.path.isdir(d):
                pending.append(d)
        if pending:
            print('dry-run PENDING missing files/dirs:', flush=True)
            for p in pending:
                print('  MISSING %s' % p, flush=True)
            raise RuntimeError('dry-run failed: %d pending items' % len(pending))
        rt = None  # settings + files + splits validated; no torch build
        rt_like_atomic(os.path.join(args.run_dir, 'dry_run.json'),
                       {'ok': True, 'splits': {
                           s: len(list_split(settings['data_root'], s))
                           for s in ('train', 'val', 'test')}})
        log('dry-run OK: settings/files/splits validated (no mmcv/GPU)')
        return

    import runtime as rt
    rt.inject_repo(settings['repo'])
    settings['_fingerprint'] = config_fingerprint(settings)

    # ---- ledger: budget_ledger.json via ops API ----
    ledger = ops.load_ledger(args.run_dir)      # from budget_ledger.json
    deadline_dt = ops.deadline_of(ledger)       # aware datetime
    global_deadline = deadline_dt.timestamp()
    log('global deadline: %s (%.2fh remaining)' %
        (deadline_dt.isoformat(), (global_deadline - time.time()) / 3600.0))

    st = load_protocol_state(args.run_dir, settings)
    ident = ensure_identity(args.run_dir, st, settings)
    save_protocol_state(args.run_dir, st)

    data_root = settings['data_root']
    assignment = rt.audit_split(data_root)
    split_hashes = {'config': settings['_hashes']['config'],
                    'split': rt._arr_sha(
                        np.array(sorted(assignment.keys())))}
    log('split audit OK')
    audit_labels(data_root, 'train', [f for f, v in assignment.items()
                                      if v['split'] == 'train'], log)

    # reserve check: 5h for final must always fit inside global deadline
    if global_deadline - time.time() < FINAL_RESERVE:
        raise ShutdownGraceful('cannot reserve final 5h')

    # =================== SHARED PREPARATION (3h total) ===================
    d_prep = mark_phase(args.run_dir, st, 'preparation')
    d_prep = min(d_prep, global_deadline)

    # reference cache (streamed to disk, never held in RAM wholesale)
    cache = rt.ReferenceCache(settings['cache_dir'], settings)
    stems = [os.path.splitext(f)[0]
             for f in list_split(data_root, 'train')]
    if not cache.complete():
        model, _ = build_student(settings)
        cache.build(model, stems, d_prep, log)
        del model
        torch.cuda.empty_cache()

    # starting D val + champion val (durations reused for pilot eval timing)
    model_d, cfg = build_student(settings)
    model_d.eval()
    t0 = time.time()
    with torch.no_grad():
        d_val, _, _ = evaluate_split_local(
            lambda i_, m_: d_inf(model_d, i_, m_), 'val', settings, log,
            d_prep, dump_counts=os.path.join(args.run_dir,
                                             'baseline_D_val_counts.json'))
    t_eval_real = time.time() - t0            # reused; no extra eval
    log('starting D val mIoU %.4f (historical baseline 42.20)' % d_val)
    del model_d
    torch.cuda.empty_cache()

    champion = rt.ChampionEnsemble(settings, settings['champion'])
    t0 = time.time()
    with torch.no_grad():
        champ_val, _, _ = evaluate_split_local(
            champion.predict, 'val', settings, log, d_prep,
            dump_counts=os.path.join(args.run_dir, 'champion_val_counts.json'))
    t_eval_real = max(t_eval_real, (time.time() - t0))
    log('champion val mIoU %.4f (historical 44.21, verified)' % champ_val)

    t_step, _ = pilot_timing(settings, args.run_dir, d_prep, log, cache,
                             reuse_eval_seconds=t_eval_real)
    T = rt.choose_steps(t_step * 1.1, t_eval_real * 1.1)
    T = max(1000, min(4000, T))
    st['chosen_T'] = T
    save_protocol_state(args.run_dir, st)
    need = 2 * T * t_step * 1.1 + 8 * t_eval_real * 1.1
    if global_deadline - time.time() < need + FINAL_RESERVE:
        raise ShutdownGraceful('insufficient time for pairs + final reserve')
    log('pilot %.3fs/step, %.1fs/eval -> T=%d' % (t_step, t_eval_real, T))

    # release champion/model objects before training; rebuild for final only
    champion = None
    torch.cuda.empty_cache()

    # =================== pair seed42 (<=10h) ===================
    d42 = min(mark_phase(args.run_dir, st, 'pair_seed42'), global_deadline)
    a_best, a_hist = train_arm(settings, args.run_dir, 'A', 42, T, d42, log,
                               cache, split_hashes)
    b_best, b_hist = train_arm(settings, args.run_dir, 'B', 42, T, d42, log,
                               cache, split_hashes)
    # gate uses the four VALIDATION-TIMEPOINT mIoUs, NOT per-class IoU
    a4 = [m for _, m in a_hist]
    b4 = [m for _, m in b_hist]
    go = ops.gate_first(a4, b4, d_val)
    log('seed42 gate: A=%s B=%s go=%s' %
        (['%.4f' % x for x in a4], ['%.4f' % x for x in b4], go))
    if not go:
        log('gate failed; no test run; retain champion; no fabricated data')
        rt_like_atomic(os.path.join(args.run_dir, 'summary.json'), {
            'baseline_D_val': d_val, 'champion_val': champ_val,
            'gate_passed': False, 'target': 47.20, 'status': 'stopped_no_gate'})
        return

    # =================== pair seed43 (<=6h) ===================
    d43 = min(mark_phase(args.run_dir, st, 'pair_seed43'), global_deadline)
    a43_hist = train_arm(settings, args.run_dir, 'A', 43, T, d43, log,
                         cache, split_hashes)[1]
    b43_hist = train_arm(settings, args.run_dir, 'B', 43, T, d43, log,
                         cache, split_hashes)[1]
    if max(m for _, m in b43_hist) <= max(m for _, m in a43_hist):
        log('seed43 reversed (B<=A); reject adoption; retain champion')
        rt_like_atomic(os.path.join(args.run_dir, 'summary.json'), {
            'baseline_D_val': d_val, 'champion_val': champ_val,
            'gate_passed': True, 'seed43_reversed': True,
            'target': 47.20, 'status': 'seed43_rejected'})
        return

    # choose best B checkpoint across seeds; tie -> seed 42, explicit file
    cands = [('B', 42, b_best, b_hist), ('B', 43,
             rt.choose_checkpoint(b43_hist), b43_hist)]
    best_seed, best = _choose_best_seed(cands)
    ckpt_path = os.path.join(args.run_dir, 'ckpt_B_s%d' % best_seed,
                             'iter_%d.pth' % best[0])
    log('best new: seed %d step %d mIoU %.4f file %s' %
        (best_seed, best[0], best[1], ckpt_path))

    # rebuild champion for the final candidate evaluation
    champion = rt.ChampionEnsemble(settings, settings['champion'])
    new_model, _ = build_student(settings)
    ck = torch.load(ckpt_path, map_location='cpu', weights_only=False)
    new_model.load_state_dict(ck['state_dict'], strict=True)
    new_model.eval()                                    # final: train False

    # final candidates on val via ONE image pass
    d_cand = min(d43, global_deadline - FINAL_RESERVE)
    with torch.no_grad():
        val_res = evaluate_candidates(settings, champion, new_model, 'val',
                                      d_cand, log)
    champ_v = val_res['champion']['miou']
    new_v = val_res['new']['miou']
    mix_v = val_res['mix']['miou']
    # ties -> champion
    if max(new_v, mix_v) <= champ_v + 1e-12:
        locked = 'champion'
    elif new_v >= mix_v:
        locked = 'new'
    else:
        locked = 'mix'
    log('locked candidate: %s (champ %.4f new %.4f mix %.4f)' %
        (locked, champ_v, new_v, mix_v))

    lock = {
        'candidate': locked,
        'chosen_checkpoint': {'path': ckpt_path,
                              'sha256': sha256(ckpt_path),
                              'step': best[0], 'miou': best[1],
                              'seed': best_seed},
        'hashes': ident['hashes'],
        'fingerprint': ident['fingerprint'],
        'split_hashes': split_hashes,
        'val_results': {'champion': champ_v, 'new': new_v, 'mix': mix_v},
        'created': time.strftime('%Y-%m-%dT%H:%M:%S'),
    }
    write_lock(args.run_dir, lock)          # BEFORE test; immutable

    # =================== final test (may start early) ===================
    final_test(settings, args.run_dir, lock, champion, new_model,
               min(global_deadline, st['phase_start'].get('final',
                   global_deadline - FINAL_RESERVE) + PHASE_CAPS['final']),
               log)
    rt_like_atomic(os.path.join(args.run_dir, 'summary.json'), {
        'baseline_D_val': d_val, 'champion_val': champ_val,
        'gate_passed': True, 'locked': locked,
        'val': {'champion': champ_v, 'new': new_v, 'mix': mix_v},
        'target': 47.20, 'status': 'tested'})


def _choose_best_seed(cands):
    """cands: list of (arm, seed, best(step,miou), hist).  Best by miou;
    ties -> seed 42."""
    best = None
    for arm, seed, bestck, hist in cands:
        if best is None or bestck[1] > best[1][1] + 1e-12:
            best = (seed, bestck)
        elif abs(bestck[1] - best[1][1]) <= 1e-12 and seed < best[0]:
            best = (seed, bestck)
    return best


def final_test(settings, run_dir, lock, champion, new_model, deadline, log):
    tag = os.path.join(run_dir, 'TEST_STARTED')
    done = os.path.join(run_dir, 'TEST_DONE')
    if os.path.isfile(done):
        log('test already complete; skip')
        return
    if os.path.isfile(tag):
        raise RuntimeError('TEST_STARTED exists without TEST_DONE: previous '
                           'test incomplete; fail closed, no automatic rerun')
    with open(tag, 'w') as f:
        f.write(time.strftime('%Y-%m-%dT%H:%M:%S'))
    try:
        with torch.no_grad():
            res = evaluate_candidates(settings, champion, new_model, 'test',
                                      deadline, log,
                                      dump_prefix=os.path.join(run_dir, 'test'))
    except ShutdownGraceful:
        raise
    except Exception:
        os.remove(tag)
        raise
    # winner stays the locked one even if another candidate scores higher
    res['locked_winner'] = lock['candidate']
    res['note'] = ('test results reported for all candidates; selection was '
                   'locked on val before the test; no claim of success')
    rt_like_atomic(os.path.join(run_dir, 'test_report.json'), res)
    log('test results: %s (locked winner: %s)' %
        ({k: round(v['miou'], 4) for k, v in res.items()
          if isinstance(v, dict) and 'miou' in v}, lock['candidate']))
    with open(done, 'w') as f:
        f.write('ok')
    os.remove(tag)


if __name__ == '__main__':
    try:
        main()
    except ShutdownGraceful as e:
        print('[shutdown-guard] %s' % e, flush=True)
        sys.exit(0)
