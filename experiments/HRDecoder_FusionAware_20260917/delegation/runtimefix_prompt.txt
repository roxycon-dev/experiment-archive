def sha256(path):
    h = hashlib.sha256()
    with open(path, 'rb') as f:
        for chunk in iter(lambda: f.read(1 << 20), b''):
            h.update(chunk)
    return h.hexdigest()

def _load_state(path):
    ckpt = torch.load(path, map_location='cpu')
    state = ckpt.get('state_dict', ckpt)
    return {k[7:] if k.startswith('module.') else k: v for k, v in state.items()}


def strict_initialize(model, student_checkpoint, reference_checkpoint):
    """Strict D2000-style init.  Accepts a full-D checkpoint (student+ref)
    or a student-only checkpoint; ``reference_checkpoint`` must be a FULL
    checkpoint whose reference tensors must be present and whose student
    tensors (when present) must be byte-equal to the student checkpoint
    (equal-B validation).  No unhashed fallback: caller hashes files first."""
    if not (os.path.isfile(student_checkpoint) and os.path.isfile(reference_checkpoint)):
        raise RuntimeError('missing checkpoint file(s); fail closed')
    s_state = _load_state(student_checkpoint)
    r_state = _load_state(reference_checkpoint)
    ref_keys = [k for k in r_state if k.startswith('reference.')]
    if not ref_keys:
        raise RuntimeError('reference checkpoint has no reference.* tensors')
    # validate ref checkpoint self-consistency: student part equals student ckpt
    s_in_r = {k: v for k, v in r_state.items() if not k.startswith('reference.')}
    for k, v in s_in_r.items():
        if k in s_state and not torch.equal(v, s_state[k]):
            raise RuntimeError('reference checkpoint student tensor differs from B: %s' % k)
    missing, unexpected = model.load_state_dict(s_state, strict=False)
    # student-only checkpoints legitimately miss reference.*; anything else fails
    bad_missing = [k for k in missing if not k.startswith('reference.')]
    if bad_missing or unexpected:
        raise RuntimeError('strict init failed; missing=%s unexpected=%s' %
                           (bad_missing, unexpected))
    if hasattr(model, 'reference'):
        ref_sd = {k[len('reference.'):]: v for k, v in r_state.items()
                  if k.startswith('reference.')}
        model.reference.load_state_dict(ref_sd, strict=True)
    for p in model.reference.parameters():
        p.requires_grad_(False)
    model.reference.eval()
    freeze_bn(model)
    return model


def freeze_bn(model):
    from torch.nn.modules.batchnorm import _BatchNorm
    for m in model.modules():
        if isinstance(m, _BatchNorm):
            m.eval()
            m.weight.requires_grad_(False)
            m.bias.requires_grad_(False)

IMG_MEAN = np.array([81.205, 50.636, 21.216], dtype=np.float32)
IMG_STD = np.array([76.252, 48.798, 21.625], dtype=np.float32)
CROP = 1024
SCALE = 2048


def paired_geometry(image, label, ref, rng):
    """image RGB HWC uint8, label HW (1..4/255), ref HWC4 float32 (cached
    native-resolution reference probability).  Deterministic under rng.
    Returns (img CHW float32 normalized, label 1HW int64, ref CHW float32,
    trace dict)."""
    import cv2
    h, w = label.shape
    image = cv2.resize(image.astype(np.float32), (SCALE, SCALE),
                       interpolation=cv2.INTER_LINEAR)
    label = cv2.resize(label.astype(np.uint8), (SCALE, SCALE),
                       interpolation=cv2.INTER_NEAREST)
    ref = cv2.resize(ref.astype(np.float32), (SCALE, SCALE),
                     interpolation=cv2.INTER_LINEAR)
    flip = bool(rng.rand() < 0.5)
    if flip:
        image = image[:, ::-1, :]
        label = label[:, ::-1]
        ref = ref[:, ::-1, :]
    y0 = int(rng.randint(0, SCALE - CROP + 1))
    x0 = int(rng.randint(0, SCALE - CROP + 1))
    image = image[y0:y0 + CROP, x0:x0 + CROP]
    label = label[y0:y0 + CROP, x0:x0 + CROP]
    ref = ref[y0:y0 + CROP, x0:x0 + CROP]
    image = (image - IMG_MEAN) / IMG_STD
    img = np.ascontiguousarray(image.transpose(2, 0, 1))
    refc = np.ascontiguousarray(ref.transpose(2, 0, 1))
    lab = label[None, :, :].astype(np.int64)
    trace = {'y0': y0, 'x0': x0, 'flip': flip, 'scale': SCALE, 'crop': CROP}
    return img, lab, refc, trace


def identity_of(img, lab, ref):
    return hashlib.sha256(np.ascontiguousarray(img).tobytes() +
                          np.ascontiguousarray(lab).tobytes() +
                          np.ascontiguousarray(ref).tobytes()).hexdigest()

def _dice(prob, target):
    """prob (C,H,W) in [0,1], target (H,W) long with 255 ignore."""
    valid = (target != 255).float()
    C = prob.shape[0]
    inter = 0.0
    card = 0.0
    for c in range(C):
        t = ((target == (c + 1)).float()) * valid
        inter = inter + (prob[c] * t).sum()
        card = card + prob[c].sum() * valid + t.sum()
    return 1.0 - (2.0 * inter + 1.0) / (card + 1.0)


def loss_pair(logits, label, ref, arm):
    """A: student dice.  B: 0.5*student dice + 0.5*fused dice where
    fused prob = 0.5*sigmoid(logits) + 0.5*ref."""
    prob_s = torch.sigmoid(logits)
    if arm == 'A':
        return _dice(prob_s, label)
    if arm != 'B':
        raise ValueError('arm must be A or B, got %r' % arm)
    prob_f = 0.5 * prob_s + 0.5 * ref
    return 0.5 * _dice(prob_s, label) + 0.5 * _dice(prob_f, label)

def choose_steps(train_seconds_per_step, eval_seconds, overhead_seconds=600):
    budget = 21600

def gate_first(a4, b4, start_miou):
    a4 = [float(x) for x in a4]
    b4 = [float(x) for x in b4]
    ma, mb = float(np.mean(a4)), float(np.mean(b4))
    paired = sum(1 for x, y in zip(a4, b4) if y > x)
    return (mb - ma >= 0.003) and paired >= 2 and mb >= start_miou


def choose_checkpoint(entries):
    """entries: list of (step, miou).  Returns (step, miou) best, ties ->
    earlier step."""
    if not entries:
        raise ValueError('empty checkpoint list')
    best = entries[0]
    for step, miou in entries[1:]:
        if miou > best[1] + 1e-12:
            best = (step, miou)
    return best

def _unwrap_img(x):

    while hasattr(x, 'data') and n < 8:

        while hasattr(x, 'data') and n < 16:

            n += 1
    return x

    while hasattr(x, 'data') and n < 8:
        x = x.data
        n += 1

        out.extend([e] if isinstance(e, dict) else [])

def per_image_counts(prob, gt):
    """prob (4,H,W) float, gt (H,W) int with 255 ignore.  Threshold 0.5,
    strict >, background (class 0) not counted."""
    counts = []
    for c in range(4):
        p = prob[c] > 0.5
        g = gt == (c + 1)
        ig = gt == 255
        tp = int(np.logical_and(p, g).sum())
        fp = int(np.logical_and(p, np.logical_and(~g, ~ig)).sum())
        fn = int(np.logical_and(~p, g).sum())
        counts.append([tp, fp, fn])
    return counts


def global_miou(all_counts):
    ious = []
    per = {}
    for c, (tp, fp, fn) in enumerate(all_counts):
        den = tp + fp + fn
        iou = tp / den if den else 0.0
        ious.append(iou)
        per['class%d' % (c + 1)] = iou
    return float(np.mean(ious)), per

def _logit(p, eps=1e-4):
    q = p.clamp(eps, 1 - eps)
    return torch.log(q / (1 - q))

    def _member_prob(self, m, img, meta, flip):
        meta_f = copy.deepcopy(meta)
        if flip:
            meta_f[0]['flip'] = True
            meta_f[0]['flip_direction'] = 'horizontal'
            x = m(return_loss=False, rescale=True, img=[img.flip(-1)],
                  img_metas=[meta_f])[0]
        else:
            x = m(return_loss=False, rescale=True, img=[img], img_metas=[meta_f])[0]
        x = np.asarray(x[0] if isinstance(x, tuple) else x, dtype=np.float32)
        p = torch.from_numpy(x).to(self.device)
        if flip:
            p = p.flip(dims=(2,))
        return p

    def predict(self, img, meta):
        """Returns (4,H,W) corrected champion probability at ori resolution."""
        with torch.no_grad():
            acc = None
            tot = 0.0
            for m, w, spec in self.members:
                p = self._member_prob(m, img, meta, False)
                acc = p * w if acc is None else acc + p * w
                p2 = self._member_prob(m, img, meta, True)
                acc = acc + p2 * w
                tot += 2.0 * w
            pf = acc / tot
            ps = self.d_model._student_prob(img, meta, True)[0]
            pr = self.d_model._reference_prob(img, meta, True)[0]
            ori = tuple(meta[0]['ori_shape'][:2])
            img_f = F.interpolate(img, size=ori, mode='bilinear',
                                  align_corners=False)[0]
            chans = []
            for c in range(4):
                cname = CLASSES[c]
                if cname in self.heads:
                    h = self.heads[cname]
                    f7 = torch.cat([img_f, _logit(pf[c:c + 1]),
                                    _logit(ps[c:c + 1]), _logit(pr[c:c + 1]),
                                    ps[c:c + 1] - pr[c:c + 1]], dim=0).unsqueeze(0)
                    newp = torch.sigmoid(_logit(pf[c:c + 1][0]) + h(f7)[0, 0])
                    chans.append(newp.unsqueeze(0))
                else:
                    chans.append(pf[c:c + 1])
            return torch.cat(chans, dim=0)


def mix_probs(p_champ, p_new):
    return 0.5 * p_champ + 0.5 * p_new

def audit_split(data_root, expected=(454, 76, 227)):
    """Verify 454/76/227 images, no dup/intersection, stems JPEG vs PNG,
    train/val/test label files exist only where expected.  Returns
    split_assignment dict filename-> {'split': ...}."""
    assignment = {}
    stems_seen = {}
    for si, split in enumerate(('train', 'val', 'test')):
        img_dir = os.path.join(data_root, 'images', split)
        lab_dir = os.path.join(data_root, 'labels', split)
        list_file = os.path.join(data_root, split + '.txt')
        imgs = sorted(f for f in os.listdir(img_dir) if f.lower().endswith(('.jpg', '.jpeg')))
        if len(imgs) != expected[si]:
            raise RuntimeError('split %s has %d images, expected %d' %
                               (split, len(imgs), expected[si]))
        if os.path.isfile(list_file):
            listed = [l.strip() for l in open(list_file, encoding='utf-8') if l.strip()]
            if sorted(os.path.splitext(x)[0] for x in listed) != \
                    sorted(os.path.splitext(x)[0] for x in imgs):
                raise RuntimeError('%s does not match %s images' % (list_file, split))
        for f in imgs:
            stem = os.path.splitext(f)[0]
            if stem in stems_seen:
                raise RuntimeError('duplicate stem across splits: %s' % stem)
            stems_seen[stem] = split
            assignment[f] = {'split': split, 'stem': stem}
        if split != 'test':
            labs = set(os.path.splitext(x)[0] for x in os.listdir(lab_dir)
                       if x.endswith('.png'))
            if labs != set(stems_seen[s] for s in stems_seen
                           if stems_seen[s] == split):
                raise RuntimeError('labels/%s mismatch' % split)
        else:
            labs = [x for x in os.listdir(lab_dir) if x.endswith('.png')]
            if labs:
                raise RuntimeError('test labels present during training — forbidden')
    return assignment

def parse_deadline(raw):
    """Accepts ISO-8601 string or epoch seconds; anything else rejected."""
    if isinstance(raw, (int, float)):
        return float(raw)
    if isinstance(raw, str):
        try:
            from datetime import datetime
            d = datetime.fromisoformat(raw.replace('Z', '+00:00'))
            if d.tzinfo is None:
                d = d.astimezone()
            return d.timestamp()
        except ValueError:
            pass
    raise ValueError('unknown deadline format (need ISO-8601 or epoch): %r' % raw)
