# fusion_core.py
"""Fusion core: paired self-ensemble training/inference helpers.

Pure torch + numpy + opencv (headless-safe). No mmcv/mmseg dependency.
Python 3.8+, CPU.

Label convention (DDR): 0 = background, 1 = EX, 2 = HE, 3 = SE, 4 = MA,
255 = ignore.  The model outputs EXACTLY 4 FOREGROUND channels
(channel i <-> label i+1); there is NO background channel.
"""
import hashlib
import math
import os

import cv2
import numpy as np
import torch
import torch.nn.functional as F
from torch.nn.modules.batchnorm import _BatchNorm

# ---------------------------------------------------------------- constants ---
IMG_MEAN = np.array([81.205, 50.636, 21.216], dtype=np.float32)
IMG_STD = np.array([76.252, 48.798, 21.625], dtype=np.float32)
IGNORE_INDEX = 255
NUM_FG = 4                       # EX, HE, SE, MA -- no background channel
REF_PREFIX = 'reference.'
SMOOTH = 1.0


# ------------------------------------------------------------------- dice ---
def dice_prob(prob, label, smooth=SMOOTH, ignore_index=IGNORE_INDEX):
    """Squared-denominator soft Dice over the 4 FOREGROUND channels.

    Channel ``i`` of ``prob`` corresponds to label ``i + 1``
    (0 = background carries no channel; 255 = ignore).

    Args:
        prob: (N, C, H, W) float probabilities in [0, 1]; C == 4.
        label: (N, 1, H, W) or (N, H, W) integer labels in 0..4 or ignore_index.
        smooth: Laplace smoothing (default 1.0).
        ignore_index: label value excluded from BOTH prediction and target.
    Returns:
        Scalar loss: mean of (1 - dice) over all N samples and C classes
        (all-ignore samples contribute 0 through the smoothing term).
    """
    assert prob.dim() == 4, 'prob must be (N, C, H, W)'
    if label.dim() == prob.dim():
        label = label.squeeze(1)
    assert label.dim() == 3, 'label must be (N, 1, H, W) or (N, H, W)'
    expected = (prob.shape[0], prob.shape[2], prob.shape[3])
    assert tuple(label.shape) == expected, \
        'label spatial shape %r must match prob %r' % (tuple(label.shape), expected)
    label = label.long()
    c = prob.shape[1]
    assert c == NUM_FG, 'model must output exactly %d foreground channels, got %d' % (NUM_FG, c)
    valid = label != ignore_index
    lab_valid = label[valid]
    assert bool(((lab_valid >= 0) & (lab_valid <= c)).all()), \
        'labels must lie in [0, %d] (0=background, %d fg classes) or equal %d' % (
            c, c, ignore_index)

    vf = valid.float().unsqueeze(1)                                # (N,1,H,W)
    idx = torch.arange(1, c + 1, device=prob.device).view(1, c, 1, 1)
    target = (label.unsqueeze(1) == idx).float() * vf              # (N,C,H,W)
    p = prob.float() * vf

    dims = (2, 3)
    num = 2.0 * (p * target).sum(dims) + smooth
    den = (p * p).sum(dims) + (target * target).sum(dims) + smooth
    loss = 1.0 - num / den                                         # (N, C)
    return loss.mean()


def _validate_ref(ref, probs):
    if ref.shape != probs.shape:
        raise ValueError('reference shape %r != prob shape %r' % (tuple(ref.shape), tuple(probs.shape)))
    if not torch.isfinite(ref).all():
        raise ValueError('reference contains non-finite values')
    if float(ref.min()) < 0.0 or float(ref.max()) > 1.0:
        raise ValueError('reference must be probabilities in [0, 1]')


def loss_pair(logits, label, ref, arm):
    """Paired loss on raw logits.

    arm 'A': Dice(sigmoid(logits), label).
    arm 'B': 0.5 * Dice(sigmoid(logits), label)
           + 0.5 * Dice(0.5*sigmoid(logits) + 0.5*ref.detach(), label).
    ``ref`` is validated (finite, [0,1], shape) and always detached.
    Returns a scalar tensor.
    """
    if arm not in ('A', 'B'):
        raise ValueError("arm must be 'A' or 'B', got %r" % (arm,))
    probs = torch.sigmoid(logits)
    ref = torch.as_tensor(ref)
    _validate_ref(ref, probs)
    ref = ref.detach()
    a = dice_prob(probs, label)
    if arm == 'A':
        return a
    fused = 0.5 * probs + 0.5 * ref
    return 0.5 * a + 0.5 * dice_prob(fused, label)


# ------------------------------------------------------------ crop helper ---
def foreground_crop_bbox(gt, crop_size, fg_prob=0.5, rng=None, num_fg=NUM_FG):
    """Top-left corner of a crop_size x crop_size window.

    With probability fg_prob (or when no lesion class 1..num_fg is present)
    the crop is image-uniform random.  Otherwise a uniformly chosen PRESENT
    class and a uniformly chosen pixel of it anchor the crop within the valid
    offset range guaranteeing pixel inclusion.
    """
    rng = np.random if rng is None else rng
    h, w = gt.shape[:2]
    cs = int(crop_size)
    cs_h, cs_w = min(cs, h), min(cs, w)

    def uniform():
        return rng.randint(0, h - cs_h + 1), rng.randint(0, w - cs_w + 1)

    present = [c for c in range(1, num_fg + 1) if (gt == c).any()]
    if not present or rng.random() < fg_prob:
        return uniform()
    cls = present[rng.randint(0, len(present))]
    ys, xs = np.where(gt == cls)
    idx = rng.randint(0, len(ys))
    py, px = int(ys[idx]), int(xs[idx])
    y_lo, y_hi = max(0, py - cs_h + 1), min(py, h - cs_h)
    x_lo, x_hi = max(0, px - cs_w + 1), min(px, w - cs_w)
    return rng.randint(y_lo, y_hi + 1), rng.randint(x_lo, x_hi + 1)


# --------------------------------------------------------------- geometry ---
def _resize_prob_hwc(prob, size_hw):
    """Resize (H, W, C) float probability map with bilinear interpolation."""
    out = np.empty((size_hw[0], size_hw[1], prob.shape[2]), dtype=np.float32)
    for c in range(prob.shape[2]):
        out[:, :, c] = cv2.resize(prob[:, :, c].astype(np.float32),
                                  (size_hw[1], size_hw[0]),
                                  interpolation=cv2.INTER_LINEAR)
    return out


def paired_geometry(image, label, reference, rng):
    """Paired geometric augmentation of image / label / reference.

    Pipeline: bilinear resize to 2048x2048 -> random rotation by a shared
    matrix (angle uniform in [-45, 45]; image and reference bilinear with 0
    border, label nearest with 255 border) -> 1024x1024 foreground-aware
    crop -> horizontal flip (p=0.5) -> image normalization.

    Args:
        image: (H, W, 3) uint8 RGB.
        label: (H, W) integer.
        reference: (H, W, 4) float probabilities in [0, 1].
        rng: np.random.RandomState.
    Returns:
        image (3, 1024, 1024) float32 normalized, label (1, 1024, 1024)
        int64, reference (4, 1024, 1024) float32, trace dict.
    """
    assert isinstance(rng, np.random.RandomState), 'rng must be np.random.RandomState'
    image = np.asarray(image)
    label = np.asarray(label)
    reference = np.asarray(reference, dtype=np.float32)
    assert image.ndim == 3 and image.shape[2] == 3 and image.dtype == np.uint8, 'image must be HWC uint8 RGB'
    assert label.ndim == 2 and label.shape == image.shape[:2], 'label must be HW matching image'
    assert reference.shape == label.shape + (NUM_FG,), 'reference must be HWC with 4 channels'
    assert reference.min() >= 0.0 and reference.max() <= 1.0, 'reference must be in [0, 1]'
    assert np.isfinite(reference).all(), 'reference must be finite'

    size = (2048, 2048)
    img_r = cv2.resize(image, (size[1], size[0]), interpolation=cv2.INTER_LINEAR)
    lab_r = cv2.resize(label.astype(np.uint8), (size[1], size[0]),
                       interpolation=cv2.INTER_NEAREST)
    ref_r = _resize_prob_hwc(reference, size)

    angle = float(rng.uniform(-45.0, 45.0))
    matrix = cv2.getRotationMatrix2D((size[1] / 2.0, size[0] / 2.0), angle, 1.0)
    img_r = cv2.warpAffine(img_r, matrix, (size[1], size[0]), flags=cv2.INTER_LINEAR,
                           borderMode=cv2.BORDER_CONSTANT, borderValue=0)
    lab_r = cv2.warpAffine(lab_r, matrix, (size[1], size[0]), flags=cv2.INTER_NEAREST,
                           borderMode=cv2.BORDER_CONSTANT, borderValue=IGNORE_INDEX)
    for c in range(NUM_FG):
        ref_r[:, :, c] = cv2.warpAffine(ref_r[:, :, c], matrix, (size[1], size[0]),
                                        flags=cv2.INTER_LINEAR,
                                        borderMode=cv2.BORDER_CONSTANT, borderValue=0.0)

    cs = 1024
    y0, x0 = foreground_crop_bbox(lab_r, cs, fg_prob=0.5, rng=rng, num_fg=NUM_FG)
    img_c = img_r[y0:y0 + cs, x0:x0 + cs]
    lab_c = lab_r[y0:y0 + cs, x0:x0 + cs]
    ref_c = ref_r[y0:y0 + cs, x0:x0 + cs]

    flip = bool(rng.random() < 0.5)
    if flip:
        img_c = img_c[:, ::-1, :]
        lab_c = lab_c[:, ::-1]
        ref_c = ref_c[:, ::-1, :]

    img_c = (img_c.astype(np.float32) - IMG_MEAN) / IMG_STD
    image_out = np.ascontiguousarray(img_c.transpose(2, 0, 1))          # (3,H,W)
    label_out = np.ascontiguousarray(lab_c.astype(np.int64))[None]       # (1,H,W)
    ref_out = np.ascontiguousarray(ref_c.astype(np.float32).transpose(2, 0, 1))

    trace = {
        'angle': angle,
        'y0': int(y0),
        'x0': int(x0),
        'flip': flip,
        'ignore_pixels': int((lab_c == IGNORE_INDEX).sum()),
        'crop': cs,
        'resize': int(size[0]),
    }
    return image_out, label_out, ref_out, trace


# ------------------------------------------------------------ model utils ---
def freeze_bn(model):
    """Put every BatchNorm layer in eval mode.

    This freezes ONLY the running statistics (buffers).  Affine parameters
    of BatchNorm layers inside allowed modules (decode_head.*, backbone.stage4.*)
    remain trainable; buffers are never parameters.
    """
    for m in model.modules():
        if isinstance(m, _BatchNorm):
            m.eval()


def configure_trainable(model):
    """Only ``decode_head.*`` and ``backbone.stage4.*`` stay trainable.

    All BatchNorm layers are switched to eval mode, so their running
    statistics (buffers) are frozen everywhere, including any reference
    submodule.  BN affine parameters stay trainable iff their owning module
    path starts with decode_head. / backbone.stage4.  Every parameter of a
    ``model.reference`` submodule is frozen.  Buffers are never parameters
    and never appear in the returned list.  Returns the sorted list of
    trainable parameter names.
    """
    trainable = []
    for name, p in model.named_parameters():
        ok = name.startswith('decode_head.') or name.startswith('backbone.stage4.')
        if name.startswith(REF_PREFIX):
            ok = False
        p.requires_grad_(ok)
        if ok:
            trainable.append(name)
    freeze_bn(model)
    if hasattr(model, 'reference'):
        for p in model.reference.parameters():
            p.requires_grad_(False)
        model.reference.eval()
    return sorted(trainable)

def sha256(path):
    """Hex sha256 digest of a file, streamed in 1 MiB chunks."""
    h = hashlib.sha256()
    with open(path, 'rb') as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b''):
            h.update(chunk)
    return h.hexdigest()


def _strip_module_prefix(state):
    return {k[7:] if k.startswith('module.') else k: v for k, v in state.items()}


def _torch_load_trusted(path):
    """torch.load of a trusted, locally produced checkpoint file.

    Uses weights_only=False where the kwarg exists; falls back for older
    torch versions without it.  Accepts str / bytes / os.PathLike.
    """
    if not isinstance(path, (str, bytes, os.PathLike)):
        raise ValueError('expected a trusted checkpoint path or state dict, got %r'
                         % (type(path).__name__,))
    try:
        return torch.load(path, map_location='cpu', weights_only=False)
    except TypeError:                       # older torch: no weights_only kwarg
        return torch.load(path, map_location='cpu')


def _load_trusted(obj):
    """Normalize a trusted checkpoint source into a stripped tensor dict.

    Accepts a bare state dict, a dict with 'state_dict', or a trusted path
    (str / bytes / os.PathLike).  Returns the 'module.'-stripped state dict.
    """
    if isinstance(obj, (str, bytes, os.PathLike)):
        obj = _torch_load_trusted(obj)
    if isinstance(obj, dict) and 'state_dict' in obj and all(
            not isinstance(v, torch.Tensor) for v in obj.values()):
        obj = obj['state_dict']
    if not isinstance(obj, dict) or not obj or not all(
            isinstance(v, torch.Tensor) for v in obj.values()):
        raise ValueError('checkpoint must be a trusted torch state_dict or bare tensor dict')
    return _strip_module_prefix(obj)


def _states_equal(a, b):
    if set(a.keys()) != set(b.keys()):
        return False
    for k in a:
        if a[k].shape != b[k].shape or a[k].dtype != b[k].dtype:
            return False
        if not torch.equal(a[k].cpu(), b[k].cpu()):
            return False
    return True


def strict_initialize(model, student_checkpoint, reference_checkpoint=None):
    """Strict checkpoint loading with reference.* partitioning.

    A FULL checkpoint contains student keys (backbone.*, decode_head.*,
    detail_branch.*, detail_head.*, detail_scale, ...) PLUS ``reference.*``
    keys.  A STUDENT-ONLY checkpoint contains no ``reference.*`` keys.
    'module.' prefixes are stripped from incoming keys.

    Procedure (NO model mutation before ALL validations pass):
      1. Partition the incoming state into student keys and embedded
         reference keys (by the ``reference.`` prefix).
      2. Validate student keys EXACTLY against the model's non-reference
         keys (no missing, no unexpected, matching shapes).
      3. If checkpoint B is given, compare the embedded reference (when
         present) against it BEFORE any load; a mismatch is rejected.
      4. Validate reference keys exactly against the model's reference.*
         keys and shapes.
      5. Merge student + ``reference.``-prefixed reference and
         ``model.load_state_dict(merged, strict=True)``.

    A model carrying reference.* keys requires a reference from somewhere
    (embedded or checkpoint B); otherwise loading is rejected.
    Returns {'student': sd, 'reference': sd_or_None} of what was loaded.
    """
    raw = _load_trusted(student_checkpoint)
    student = {k: v for k, v in raw.items() if not k.startswith(REF_PREFIX)}
    embedded_ref = {k[len(REF_PREFIX):]: v for k, v in raw.items()
                    if k.startswith(REF_PREFIX)}

    model_state = model.state_dict()
    student_expected = {k for k in model_state if not k.startswith(REF_PREFIX)}
    ref_expected = {k[len(REF_PREFIX):] for k in model_state if k.startswith(REF_PREFIX)}

    # ---- student key / shape validation (no mutation yet) ----
    missing = student_expected - set(student)
    unexpected = set(student) - student_expected
    if missing or unexpected:
        raise ValueError('strict student mismatch: missing=%d %s unexpected=%d %s' % (
            len(missing), sorted(missing)[:5], len(unexpected), sorted(unexpected)[:5]))
    for k in sorted(student):
        if tuple(student[k].shape) != tuple(model_state[k].shape):
            raise ValueError('strict shape mismatch for %r: ckpt %r vs model %r' % (
                k, tuple(student[k].shape), tuple(model_state[k].shape)))

    # ---- reference resolution and cross-check (still no mutation) ----
    reference = None
    if reference_checkpoint is not None:
        reference = _load_trusted(reference_checkpoint)
    if embedded_ref:
        if reference is not None and not _states_equal(embedded_ref, reference):
            raise ValueError('full-D embedded reference differs from checkpoint B '
                             'reference; rejected')
        ref_to_load = embedded_ref
    else:
        ref_to_load = reference
    if ref_to_load is not None:
        r_missing = ref_expected - set(ref_to_load)
        r_unexpected = set(ref_to_load) - ref_expected
        if r_missing or r_unexpected:
            raise ValueError('strict reference mismatch: missing=%d %s unexpected=%d %s' % (
                len(r_missing), sorted(r_missing)[:5],
                len(r_unexpected), sorted(r_unexpected)[:5]))
        for k in sorted(ref_to_load):
            if tuple(ref_to_load[k].shape) != tuple(model_state[REF_PREFIX + k].shape):
                raise ValueError('strict reference shape mismatch for %r' % (k,))
    elif ref_expected:
        raise ValueError('model contains reference.* keys but no reference was '
                         'provided (neither embedded nor checkpoint B); rejected')

    # ---- ALL validations passed: mutate the model exactly once ----
    merged = dict(student)
    merged.update({REF_PREFIX + k: v for k, v in ref_to_load.items()} if ref_to_load else {})
    model.load_state_dict(merged, strict=True)

    return {'student': student, 'reference': ref_to_load}

# ------------------------------------------------------------------ toy model ---
class ToyStage(nn.Module):
    def __init__(self, cin, cout):
        super().__init__()
        self.conv = nn.Conv2d(cin, cout, 3, padding=1)
        self.bn = nn.BatchNorm2d(cout)
        self.relu = nn.ReLU(inplace=True)

    def forward(self, x):
        return self.relu(self.bn(self.conv(x)))


class ToyBackbone(nn.Module):
    def __init__(self):
        super().__init__()
        self.stage1 = ToyStage(3, 8)
        self.stage2 = ToyStage(8, 8)
        self.stage3 = ToyStage(8, 8)
        self.stage4 = ToyStage(8, 8)

    def forward(self, x):
        return self.stage4(self.stage3(self.stage2(self.stage1(x))))


class ToyDecodeHead(nn.Module):
    def __init__(self):
        super().__init__()
        self.conv = nn.Conv2d(8, 4, 3, padding=1)
        self.bn = nn.BatchNorm2d(4)

    def forward(self, x):
        return self.bn(self.conv(x))


class ToyReference(nn.Module):
    """Reference submodule: contributes reference.backbone.*, reference.decode_head.*,
    reference.detail_scale to the model state dict."""

    def __init__(self):
        super().__init__()
        self.backbone = ToyBackbone()
        self.decode_head = ToyDecodeHead()
        self.register_buffer('detail_scale', torch.ones(1))


class ToyModel(nn.Module):
    """Toy stand-in for the real D model: student keys (backbone.*, decode_head.*,
    detail_branch.*, detail_head.*, detail_scale) PLUS reference.* keys."""

    def __init__(self):
        super().__init__()
        self.backbone = ToyBackbone()
        self.decode_head = ToyDecodeHead()
        self.detail_branch = ToyStage(3, 8)
        self.detail_head = nn.Conv2d(8, 4, 1)
        self.register_buffer('detail_scale', torch.ones(1))
        self.reference = ToyReference()

    def forward(self, x):
        return self.decode_head(self.backbone(x))

def _manual_dice(prob, label, smooth=1.0, ignore=255):
    """Manual squared-Dice: channel k <-> label k+1; ignore excluded."""
    n, c = prob.shape[0], prob.shape[1]
    vals = []
    for i in range(n):
        m = label[i] != ignore
        for k in range(c):
            p = prob[i, k][m]
            t = (label[i][m] == (k + 1)).float()
            vals.append(1.0 - (2 * (p * t).sum() + smooth) /
                        ((p * p).sum() + (t * t).sum() + smooth))
    return float(np.mean(vals))


class TestDice(unittest.TestCase):
    def test_label_shape_nhw_against_nchw_prob(self):
        # regression: label (N,H,W) must be accepted against (N,C,H,W) prob
        torch.manual_seed(1)
        prob = torch.rand(3, 4, 16, 16)
        label = torch.randint(0, 5, (3, 16, 16))
        label[0, 0, 0] = 255
        label[1] = 255  # all-ignore sample
        self.assertAlmostEqual(dice_prob(prob, label).item(),
                               _manual_dice(prob, label), places=6)

    def test_numeric_parity_vs_manual_incl_class4_and_background(self):
        torch.manual_seed(2)
        prob = torch.rand(2, 4, 12, 12)
        label = torch.zeros(2, 12, 12, dtype=torch.long)   # background everywhere
        label[0, 2:6, 2:6] = 4                              # MA present
        label[0, 7:9, 7:9] = 1                              # EX present
        label[1, 0, 0] = 255
        self.assertAlmostEqual(dice_prob(prob, label).item(),
                               _manual_dice(prob, label), places=6)

    def test_all_background_is_zero(self):
        prob = torch.rand(2, 4, 8, 8)
        label = torch.zeros(2, 8, 8, dtype=torch.long)      # background only
        self.assertAlmostEqual(dice_prob(prob, label).item(), 0.0, places=7)

    def test_all_ignore_loss_zero_graph_connected_gradients_zero(self):
        prob = torch.rand(1, 4, 8, 8, requires_grad=True)
        label = torch.full((1, 8, 8), 255, dtype=torch.long)
        loss = dice_prob(prob, label)
        self.assertAlmostEqual(loss.item(), 0.0, places=7)
        loss.backward()                                     # graph must be connected
        self.assertIsNotNone(prob.grad)
        self.assertTrue(torch.isfinite(prob.grad).all())
        self.assertEqual(float(prob.grad.abs().sum()), 0.0)

    def test_ignores_label_255_only_through_ignore_index(self):
        torch.manual_seed(3)
        prob = torch.rand(1, 4, 8, 8)
        label = torch.randint(0, 5, (1, 8, 8))
        label_masked = label.clone()
        label_masked[0, 0, 0] = 255
        self.assertLess(dice_prob(prob, label).item(), 1.0)

    def test_accepts_label_with_channel_dim(self):
        torch.manual_seed(4)
        prob = torch.rand(2, 4, 8, 8)
        label = torch.randint(0, 5, (2, 1, 8, 8))
        self.assertAlmostEqual(dice_prob(prob, label).item(),
                               dice_prob(prob, label.squeeze(1)).item(), places=7)

    def test_is_differentiable(self):
        torch.manual_seed(5)
        prob = torch.rand(1, 4, 8, 8, requires_grad=True)
        label = torch.randint(0, 5, (1, 8, 8))
        dice_prob(prob, label).backward()
        self.assertIsNotNone(prob.grad)
        self.assertTrue(torch.isfinite(prob.grad).all())

# -------------------------------------------------------------- loss_pair ---
class TestLossPair(unittest.TestCase):
    def test_arm_a_matches_plain_dice(self):
        torch.manual_seed(6)
        logits = torch.randn(2, 4, 8, 8)
        label = torch.randint(0, 5, (2, 8, 8))
        ref = torch.rand(2, 4, 8, 8)
        a = loss_pair(logits, label, ref, 'A')
        self.assertAlmostEqual(a.item(),
                               dice_prob(torch.sigmoid(logits), label).item(), places=6)

    def test_arm_b_composition(self):
        torch.manual_seed(7)
        logits = torch.randn(1, 4, 8, 8)
        label = torch.randint(0, 5, (1, 8, 8))
        ref = torch.rand(1, 4, 8, 8)
        b = loss_pair(logits, label, ref, 'B')
        probs = torch.sigmoid(logits)
        expect = 0.5 * dice_prob(probs, label) + \
            0.5 * dice_prob(0.5 * probs + 0.5 * ref, label)
        self.assertAlmostEqual(b.item(), expect.item(), places=6)

    def test_reference_is_detached(self):
        torch.manual_seed(8)
        logits = torch.randn(1, 4, 8, 8)
        label = torch.randint(0, 5, (1, 8, 8))
        ref = torch.rand(1, 4, 8, 8, requires_grad=True)
        loss_pair(logits, label, ref, 'B').backward()
        self.assertIsNone(ref.grad)

    def test_reference_validation(self):
        logits = torch.randn(1, 4, 8, 8)
        label = torch.randint(0, 5, (1, 8, 8))
        with self.assertRaises(ValueError):
            loss_pair(logits, label, torch.rand(1, 4, 4, 4), 'B')          # shape
        bad = torch.rand(1, 4, 8, 8)
        bad[0, 0, 0, 0] = float('nan')
        with self.assertRaises(ValueError):
            loss_pair(logits, label, bad, 'B')                              # non-finite
        with self.assertRaises(ValueError):
            loss_pair(logits, label, torch.rand(1, 4, 8, 8) * 2.0, 'B')     # out of range
        with self.assertRaises(ValueError):
            loss_pair(logits, label, torch.rand(1, 4, 8, 8), 'C')           # arm

class TestTrainable(unittest.TestCase):
    def test_only_authorized_params_trainable(self):
        model, names = _prepared_model()
        for n, p in model.named_parameters():
            if n.startswith(('decode_head.', 'backbone.stage4.')):
                self.assertTrue(p.requires_grad, n)
            else:
                self.assertFalse(p.requires_grad, n)
        self.assertTrue(all(n.startswith(('decode_head.', 'backbone.stage4.'))
                            for n in names))

    def test_bn_affine_in_allowed_modules_trainable_but_stats_frozen(self):
        model, _ = _prepared_model()
        # BN affine params inside decode_head/backbone.stage4 remain trainable
        for n, p in model.named_parameters():
            if 'bn' in n and n.startswith(('decode_head.', 'backbone.stage4.')):
                self.assertTrue(p.requires_grad, n)
        # and running stats stay untouched after optimizer updates
        opt = torch.optim.SGD([p for p in model.parameters() if p.requires_grad], lr=0.1)
        bn_stats = {mn: (m.running_mean.clone(), m.running_var.clone())
                    for mn, m in model.named_modules() if isinstance(m, nn.BatchNorm2d)}
        img, label = _batch()
        loss = loss_pair(model(img), label, torch.rand(2, 4, 32, 32), 'B')
        opt.zero_grad()
        loss.backward()
        opt.step()
        for mn, m in model.named_modules():
            if isinstance(m, nn.BatchNorm2d):
                self.assertTrue(torch.equal(m.running_mean, bn_stats[mn][0]), mn)
                self.assertTrue(torch.equal(m.running_var, bn_stats[mn][1]), mn)

    def test_gradients_and_updates_only_authorized(self):
        model, _ = _prepared_model()
        opt = torch.optim.SGD([p for p in model.parameters() if p.requires_grad], lr=0.1)
        before = {n: p.detach().clone() for n, p in model.named_parameters()}
        img, label = _batch()
        loss = loss_pair(model(img), label, torch.rand(2, 4, 32, 32), 'B')
        opt.zero_grad()
        loss.backward()
        auth_grad = [n for n, p in model.named_parameters()
                     if p.grad is not None and p.grad.abs().sum() > 0]
        self.assertTrue(auth_grad)
        self.assertTrue(all(n.startswith(('decode_head.', 'backbone.stage4.'))
                            for n in auth_grad))
        opt.step()
        changed = [n for n, p in model.named_parameters()
                   if not torch.equal(before[n], p.detach())]
        self.assertTrue(changed)
        self.assertTrue(all(n.startswith(('decode_head.', 'backbone.stage4.'))
                            for n in changed))

    def test_bn_eval_even_in_train_mode(self):
        model, _ = _prepared_model()
        model.train()
        for m in model.modules():
            if isinstance(m, nn.BatchNorm2d):
                self.assertFalse(m.training)
        model.reference.train()
        self.assertFalse(any(m.training for m in model.reference.modules()
                             if isinstance(m, nn.BatchNorm2d)))

    def test_freeze_bn_covers_all(self):
        model = ToyModel().train()
        freeze_bn(model)
        self.assertFalse(any(m.training for m in model.modules()
                             if isinstance(m, nn.BatchNorm2d)))

class TestStrictInitialize(unittest.TestCase):

        return _flat_ckpt(model.state_dict())          # includes reference.* keys

        self.assertFalse(torch.equal(model2.decode_head.conv.weight,

def _marker_scene(size=512):
    img = np.zeros((size, size, 3), dtype=np.uint8)
    label = np.zeros((size, size), dtype=np.int64)
    y0, y1, x0, x1 = 100, 220, 130, 300
    img[y0:y1, x0:x1] = 200
    label[y0:y1, x0:x1] = 3
    reference = np.eye(4, dtype=np.float32)[label]     # (H, W, 4) one-hot
    return img, label, reference


def _interior_mask(mask, k=3):
    m = mask.copy()
    for _ in range(k):
        m = m & np.roll(m, 1, 0) & np.roll(m, -1, 0) & np.roll(m, 1, 1) & np.roll(m, -1, 1)
    return m


class TestPairedGeometry(unittest.TestCase):
    def test_marker_alignment(self):
        img, label, reference = _marker_scene()
        out_img, out_label, out_ref, trace = paired_geometry(
            img, label, reference, np.random.RandomState(7))
        self.assertEqual(out_img.shape, (3, 1024, 1024))
        self.assertEqual(out_label.shape, (1, 1024, 1024))
        self.assertEqual(out_label.dtype, np.int64)
        self.assertEqual(out_ref.shape, (4, 1024, 1024))
        self.assertEqual(out_ref.dtype, np.float32)
        lab = out_label[0]
        cls3 = (lab == 3)
        self.assertTrue(cls3.any(), 'rotated marker class must survive crop')
        interior = _interior_mask(cls3)
        self.assertTrue(interior.any())
        ref_c = out_ref
        self.assertTrue((ref_c[3][interior] > 0.99).all())
        others = np.concatenate([ref_c[0][interior], ref_c[1][interior], ref_c[2][interior]])
        self.assertTrue((others < 0.01).all())
        border = (lab == 255)
        if border.any():
            self.assertLessEqual(float(ref_c.max(axis=0)[border].max()), 1e-6)
        self.assertEqual(trace['crop'], 1024)
        self.assertEqual(trace['resize'], 2048)
        self.assertTrue(-45.0 <= trace['angle'] <= 45.0)
        img_again, *_ = paired_geometry(img, label, reference, np.random.RandomState(7))
        self.assertTrue(np.array_equal(img_again, out_img))

    def test_seed_reproducibility(self):
        img, label, reference = _marker_scene()
        a = paired_geometry(img, label, reference, np.random.RandomState(11))
        b = paired_geometry(img, label, reference, np.random.RandomState(11))
        c = paired_geometry(img, label, reference, np.random.RandomState(12))
        for x, y in zip(a[:3], b[:3]):
            self.assertTrue(np.array_equal(x, y))
        self.assertEqual(a[3], b[3])
        self.assertNotEqual(a[3], c[3])

    def test_foreground_crop_bbox_seed_trace_matches_source_semantics(self):
        # crop helper is copied verbatim from the source implementation;
        # identical seeds must reproduce identical (y0, x0) traces.
        rng_src = np.random.RandomState(3)
        gt = np.zeros((256, 256), dtype=np.int64)
        gt[40:80, 60:120] = 2
        gt[150:200, 100:180] = 4
        trace1 = [foreground_crop_bbox(gt, 64, fg_prob=0.5, rng=rng_src)
                  for _ in range(20)]
        rng_again = np.random.RandomState(3)
        trace2 = [foreground_crop_bbox(gt, 64, fg_prob=0.5, rng=rng_again)
                  for _ in range(20)]
        self.assertEqual(trace1, trace2)
        rng_other = np.random.RandomState(4)
        trace3 = [foreground_crop_bbox(gt, 64, fg_prob=0.5, rng=rng_other)
                  for _ in range(20)]
        self.assertNotEqual(trace1, trace3)

    def test_fg_anchor_guarantees_inclusion(self):
        gt = np.zeros((256, 256), dtype=np.int64)
        gt[100:110, 100:110] = 1

        img, label, _ = _marker_scene(64)
        bad = np.zeros((64, 64, 4), dtype=np.float32)
        bad[0, 0, 0] = 1.5
        with self.assertRaises(AssertionError):
            paired_geometry(img, label, bad, np.random.RandomState(0))

Key corrections applied:

1. **Dice shape check** — label `(N,H,W)` is now compared against `(N, H, W)` derived correctly from prob; the old `prob.shape[:2] + prob.shape[2:]` bug (always `(N,C,H,W)`) is gone.
2. **DDR semantics** — labels `0=background, 1=EX, 2=HE, 3=SE, 4=MA, 255=ignore`; targets built as `target[:,i] = (label == i+1)` over exactly 4 channels; all-background gives loss 0; all-ignore gives loss 0 with a connected graph and zero gradients.
3. **strict_initialize** — partitions incoming `reference.*` keys, validates student-only keys against the model's non-reference keys, cross-checks the embedded reference against checkpoint B **before any load**, then loads `student + reference.*` via one `load_state_dict(strict=True)`; supports `os.PathLike` with `weights_only=False` (with fallback for older torch); a model with `reference.*` keys and no reference is rejected; no mutation on any rejection (tested explicitly).
4. **BN freeze docs** — docstrings no longer claim BN affine parameters are frozen; running stats (buffers) are frozen via eval mode, affines in `decode_head.*`/`backbone.stage4.*` stay trainable (covered by a dedicated test).
5. **Geometry/crop** — `foreground_crop_bbox` retains the exact source semantics; a seed-trace reproducibility test pins the copied behavior.
6. **Tests** — converted entirely to `unittest`, including label 4 and all-background-zero cases, strict conflicting/missing/extra/shape rejections, and the geometry suite.
